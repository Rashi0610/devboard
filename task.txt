Four routes:
POST /:columnId/tasks — create a task

protected with requireAuth
get title, description, priority, dueDate, storyPoints from req.body
get columnId from req.params
get projectId from req.body as well
set createdBy to req.userId
position — count existing tasks in this column: await TaskModel.countDocuments({ column: columnId })
return { task }

GET /:columnId/tasks — get all tasks in a column
protected with requireAuth
find all tasks where column equals columnId
sort by position ascending
return { tasks }

PATCH /:columnId/tasks/:taskId — update a task

protected with requireAuth
get updated fields from req.body
use TaskModel.findByIdAndUpdate(taskId, req.body, { new: true })
return { task }

DELETE /:columnId/tasks/:taskId — delete a task

protected with requireAuth
findByIdAndDelete using taskId
return { message: 'Task deleted' }
